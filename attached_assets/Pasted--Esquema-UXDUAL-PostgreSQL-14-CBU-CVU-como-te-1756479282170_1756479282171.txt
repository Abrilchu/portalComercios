-- =========================================
-- Esquema UXDUAL (PostgreSQL 14+)
-- - CBU/CVU como texto con validación 22 dígitos
-- - Únicos en CUIT, email, DNI, CBU, CVU, alias de cuenta
-- - IDENTITY en lugar de SERIAL
-- =========================================

DROP SCHEMA IF EXISTS uxdual CASCADE;
CREATE SCHEMA uxdual;
SET search_path TO uxdual;

-- ========================
-- Tablas base
-- ========================

CREATE TABLE Estado (
  estadoId      INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(100) NOT NULL,
  descripcion   VARCHAR(255)
);

CREATE TABLE Direccion (
  direccionId   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  calle         VARCHAR(150),
  fechaCreacion DATE DEFAULT CURRENT_DATE,
  numero        INT,
  piso          VARCHAR(20),
  barrio        VARCHAR(100)
);

CREATE TABLE Comercio (
  comercioId    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(150) NOT NULL,
  cuit          VARCHAR(20)  NOT NULL UNIQUE,
  estadoId      INT REFERENCES Estado(estadoId),
  fechaCreacion DATE DEFAULT CURRENT_DATE
);

CREATE TABLE Sucursal (
  sucursalId    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  comercioId    INT NOT NULL REFERENCES Comercio(comercioId),
  direccionId   INT REFERENCES Direccion(direccionId),
  nombre        VARCHAR(150) NOT NULL,
  descripcion   VARCHAR(255)
);

CREATE TABLE CuentaBancaria (
  cuentaId      INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  sucursalId    INT NOT NULL REFERENCES Sucursal(sucursalId),
  alias         VARCHAR(50) UNIQUE,
  -- CBU/CVU como TEXTO para conservar ceros iniciales y no exceder BIGINT
  cbu           VARCHAR(22),
  cvu           VARCHAR(22),
  banco         VARCHAR(100),
  estadoId      INT REFERENCES Estado(estadoId),
  -- Validación opcional de formato de 22 dígitos
  CONSTRAINT cbu_formato_chk CHECK (cbu IS NULL OR cbu ~ '^[0-9]{22}$'),
  CONSTRAINT cvu_formato_chk CHECK (cvu IS NULL OR cvu ~ '^[0-9]{22}$'),
  CONSTRAINT cbu_unico UNIQUE (cbu),
  CONSTRAINT cvu_unico UNIQUE (cvu)
);

CREATE TABLE PuntoVta (
  puntoVtaId    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  sucursalId    INT NOT NULL REFERENCES Sucursal(sucursalId),
  estadoId      INT REFERENCES Estado(estadoId),
  descripcion   VARCHAR(255)
);

CREATE TABLE Usuario (
  empleadoId    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  sucursalId    INT REFERENCES Sucursal(sucursalId),
  nombre        VARCHAR(100) NOT NULL,
  apellido      VARCHAR(100),
  email         VARCHAR(150) UNIQUE,
  telefono      VARCHAR(50),
  estadoId      INT REFERENCES Estado(estadoId),
  fechaCreacion DATE DEFAULT CURRENT_DATE,
  dni           VARCHAR(20) UNIQUE,
  pinHash       VARCHAR(255)
);

CREATE TABLE Rol (
  rolId         INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(100) NOT NULL,
  descripcion   VARCHAR(255)
);

CREATE TABLE UserRol (
  empRolId      INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  empleadoId    INT NOT NULL REFERENCES Usuario(empleadoId),
  rolId         INT NOT NULL REFERENCES Rol(rolId),
  CONSTRAINT userrol_unico UNIQUE (empleadoId, rolId)
);

CREATE TABLE Permiso (
  permisoId     INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(100) NOT NULL,
  alias         VARCHAR(50),
  descripcion   VARCHAR(255)
);

CREATE TABLE Recurso (
  recursoId     INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(100) NOT NULL,
  alias         VARCHAR(50),
  descripcion   VARCHAR(255)
);

CREATE TABLE RolResPer (
  rolResPerId   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  rolId         INT REFERENCES Rol(rolId),
  permisoId     INT REFERENCES Permiso(permisoId),
  recursoId     INT REFERENCES Recurso(recursoId),
  CONSTRAINT rol_perm_recurso_unico UNIQUE (rolId, permisoId, recursoId)
);

CREATE TABLE Qr (
  qrId          INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  estadoId      INT REFERENCES Estado(estadoId),
  puntoVtaId    INT REFERENCES PuntoVta(puntoVtaId),
  fechaCreacion DATE DEFAULT CURRENT_DATE,
  ultimaRotacion DATE,
  tokenHash     VARCHAR(255)
);

CREATE TABLE EstadoTransaccion (
  estadoTrxId   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre        VARCHAR(100) NOT NULL,
  descripcion   VARCHAR(255)
);

CREATE TABLE Trx (
  trxId         INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  qrId          INT REFERENCES Qr(qrId),
  estadoTrxId   INT REFERENCES EstadoTransaccion(estadoTrxId),
  monto         NUMERIC(15,2),
  metodo        VARCHAR(50),
  moneda        VARCHAR(10),
  fechaCreacion DATE DEFAULT CURRENT_DATE,
  fechaAprobacion DATE,
  idempotencyKey  VARCHAR(100),
  originCUIT    VARCHAR(20),
  destinationCUIT VARCHAR(20)
);

-- ========================
-- Índices recomendados (FKs y búsquedas comunes)
-- ========================

CREATE INDEX idx_sucursal_comercio   ON Sucursal(comercioId);
CREATE INDEX idx_sucursal_direccion  ON Sucursal(direccionId);

CREATE INDEX idx_cuentab_sucursal    ON CuentaBancaria(sucursalId);
CREATE INDEX idx_puntovta_sucursal   ON PuntoVta(sucursalId);

CREATE INDEX idx_usuario_sucursal    ON Usuario(sucursalId);
CREATE INDEX idx_usuario_estado      ON Usuario(estadoId);

CREATE INDEX idx_qr_puntovta         ON Qr(puntoVtaId);
CREATE INDEX idx_qr_estado           ON Qr(estadoId);

CREATE INDEX idx_trx_qr              ON Trx(qrId);
CREATE INDEX idx_trx_estado          ON Trx(estadoTrxId);
CREATE INDEX idx_trx_fecha           ON Trx(fechaCreacion);
